import { useRouter } from "expo-router";
import React, { useEffect, useState } from "react";
import {
    FlatList,
    Text,
    TextInput,
    TouchableOpacity,
    View,
} from "react-native";
import { getAllCategories } from "../../src/db/categories.repo";
import { initDB, seedData } from "../../src/db/db";
import type { Category } from "../../src/models/types";

export default function CategoryScreen() {
  const [categories, setCategories] = useState<Category[]>([]);
  const [filtered, setFiltered] = useState<Category[]>([]);
  const [search, setSearch] = useState("");
  const router = useRouter();

  useEffect(() => {
    (async () => {
      await initDB();
      await seedData();
      const data = await getAllCategories();
      setCategories(data);
      setFiltered(data);
    })();
  }, []);

  // 📦 Xử lý lọc danh mục theo tên
  const handleSearch = (text: string) => {
    setSearch(text);
    if (text.trim() === "") {
      setFiltered(categories);
    } else {
      const filteredData = categories.filter((item) =>
        item.name.toLowerCase().includes(text.toLowerCase())
      );
      setFiltered(filteredData);
    }
  };

  return (
    <View style={{ flex: 1, backgroundColor: "#f8f9fa", padding: 16 }}>
      <Text
        style={{
          fontSize: 24,
          fontWeight: "bold",
          marginBottom: 12,
          textAlign: "center",
        }}
      >
        Danh mục sản phẩm
      </Text>

      {/* 🔍 Ô tìm kiếm */}
      <TextInput
        placeholder="Tìm danh mục..."
        value={search}
        onChangeText={handleSearch}
        style={{
          backgroundColor: "white",
          borderRadius: 8,
          padding: 10,
          marginBottom: 16,
          borderWidth: 1,
          borderColor: "#ddd",
        }}
      />

      {/* 📋 Danh sách danh mục */}
      <FlatList
        data={filtered}
        keyExtractor={(item) => item.category_id.toString()}
        renderItem={({ item }) => (
          <TouchableOpacity
            onPress={() =>
              router.push({
                pathname: "/product",
                params: {
                  category_id: item.category_id.toString(),
                  category_name: item.name,
                },
              })
            }
          >
            <View
              style={{
                backgroundColor: "white",
                padding: 14,
                borderRadius: 10,
                marginBottom: 10,
                shadowColor: "#000",
                shadowOpacity: 0.1,
                shadowRadius: 4,
                elevation: 2,
              }}
            >
              <Text style={{ fontWeight: "600", fontSize: 16 }}>
                {item.name}
              </Text>
              <Text style={{ color: "#777", marginTop: 4 }}>
                Ngày cập nhật: {item.updated_at}
              </Text>
            </View>
          </TouchableOpacity>
        )}
      />
    </View>
  );
}

// app/product.tsx
import { useLocalSearchParams } from "expo-router";
import React, { useEffect, useState } from "react";
import { Alert, FlatList, Text, TouchableOpacity, View } from "react-native";
import { initDB, seedData } from "../../src/db/db";
import {
    deleteProduct,
    getAllProducts,
    getProductsByCategory,
    updateProduct,
} from "../../src/db/product.repo";
import type { Product } from "../../src/models/types";

export default function ProductScreen() {
  const [products, setProducts] = useState<Product[]>([]);
  const { category_id, category_name } = useLocalSearchParams();

  // 🟢 Load dữ liệu ban đầu
  useEffect(() => {
    (async () => {
      await initDB();
      await seedData();

      let list: Product[] = [];
      if (category_id) {
        list = await getProductsByCategory(Number(category_id));
      } else {
        list = await getAllProducts();
      }
      setProducts(list);
    })();
  }, [category_id]);

  // 🗑️ Hàm xóa sản phẩm
  const handleDelete = async (id: number) => {
    Alert.alert("Xác nhận", "Bạn có chắc muốn xóa sản phẩm này?", [
      { text: "Hủy", style: "cancel" },
      {
        text: "Xóa",
        style: "destructive",
        onPress: async () => {
          await deleteProduct(id);
          const list = category_id
            ? await getProductsByCategory(Number(category_id))
            : await getAllProducts();
          setProducts(list);
        },
      },
    ]);
  };

  // ✏️ Hàm chỉnh sửa giá sản phẩm (ví dụ tăng thêm 10.000₫)
  const handleEdit = async (product: Product) => {
    const newPrice = product.price + 10000;
    await updateProduct({ ...product, price: newPrice });
    const list = category_id
      ? await getProductsByCategory(Number(category_id))
      : await getAllProducts();
    setProducts(list);
  };

  return (
    <View style={{ flex: 1, backgroundColor: "#f8f9fa", padding: 16 }}>
      <Text
        style={{
          fontSize: 22,
          fontWeight: "bold",
          marginBottom: 12,
          textAlign: "center",
        }}
      >
        {category_name ? `Sản phẩm: ${category_name}` : "Tất cả sản phẩm"}
      </Text>

      <FlatList
        data={products}
        keyExtractor={(item) => item.product_id.toString()}
        renderItem={({ item }) => (
          <View
            style={{
              backgroundColor: "white",
              padding: 12,
              borderRadius: 10,
              marginBottom: 10,
              flexDirection: "row",
              justifyContent: "space-between",
              alignItems: "center",
              shadowColor: "#000",
              shadowOpacity: 0.1,
              shadowRadius: 4,
              elevation: 2,
            }}
          >
            {/* Thông tin sản phẩm */}
            <View>
              <Text style={{ fontWeight: "600", fontSize: 16 }}>
                {item.name}
              </Text>
              <Text style={{ color: "#1e88e5", marginTop: 4 }}>
                {item.price.toLocaleString()}₫
              </Text>
              {item.category_name && (
                <Text style={{ color: "#777", marginTop: 2 }}>
                  {item.category_name}
                </Text>
              )}
            </View>

            {/* Nút sửa / xóa */}
            <View style={{ flexDirection: "row", gap: 8 }}>
              <TouchableOpacity
                style={{
                  backgroundColor: "#1976d2",
                  paddingVertical: 6,
                  paddingHorizontal: 10,
                  borderRadius: 8,
                }}
                onPress={() => handleEdit(item)}
              >
                <Text style={{ color: "white", fontWeight: "bold" }}>Sửa</Text>
              </TouchableOpacity>

              <TouchableOpacity
                style={{
                  backgroundColor: "#e53935",
                  paddingVertical: 6,
                  paddingHorizontal: 10,
                  borderRadius: 8,
                }}
                onPress={() => handleDelete(item.product_id)}
              >
                <Text style={{ color: "white", fontWeight: "bold" }}>Xóa</Text>
              </TouchableOpacity>
            </View>
          </View>
        )}
      />
    </View>
  );
}

export async function initDB() {
  await db.execAsync(`
    -- Bảng danh mục sản phẩm
    CREATE TABLE IF NOT EXISTS categories (
      category_id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      remote_id TEXT,
      updated_at TEXT NOT NULL DEFAULT (datetime('now'))
    );

    -- Bảng sản phẩm
    CREATE TABLE IF NOT EXISTS products (
      product_id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      price REAL NOT NULL CHECK(price >= 0),
      description TEXT,
      image_uri TEXT,
      category_id INTEGER,
      FOREIGN KEY (category_id) REFERENCES categories(category_id) ON DELETE SET NULL
    );
  `);
}

import db from "../db/db";
import { Category } from "../models/types";

// ➕ Thêm danh mục mới
export async function addCategory(name: string, remote_id?: string) {
  const { lastInsertRowId } = await db.runAsync(
    `INSERT INTO categories (name, remote_id, updated_at)
     VALUES (?, ?, datetime('now'))`,
    [name, remote_id || null]
  );
  return lastInsertRowId;
}

// ✏️ Cập nhật danh mục
export async function updateCategory(
  id: number,
  name: string,
  remote_id?: string
) {
  await db.runAsync(
    `UPDATE categories
     SET name = ?, remote_id = ?, updated_at = datetime('now')
     WHERE category_id = ?`,
    [name, remote_id || null, id]
  );
  return true;
}

// 🗑️ Xóa danh mục
export async function deleteCategory(id: number) {
  await db.runAsync(`DELETE FROM categories WHERE category_id = ?`, [id]);
  return true;
}

export async function getAllCategories(): Promise<Category[]> {
  return await db.getAllAsync<Category>(
    `SELECT * FROM categories ORDER BY name COLLATE NOCASE ASC`
  );
}

export async function getCategoryById(id: number): Promise<Category | null> {
  return await db.getFirstAsync<Category>(
    `SELECT * FROM categories WHERE category_id = ?`, [id]
  );
}

import db from "../db/db";
import { Product } from "../models/types";

// 🧾 Lấy tất cả sản phẩm
export async function getAllProducts(): Promise<Product[]> {
  return await db.getAllAsync<Product>(`
    SELECT p.*, c.name AS category_name
    FROM products p
    LEFT JOIN categories c ON p.category_id = c.category_id
    ORDER BY p.product_id DESC
  `);
}

// 🔍 Tìm sản phẩm theo id
export async function getProductById(id: number): Promise<Product | null> {
  return await db.getFirstAsync<Product>(
    `SELECT * FROM products WHERE product_id = ?`, [id]
  );
}

// 🔍 Tìm sản phẩm theo tên
export async function findProductsByName(name: string): Promise<Product[]> {
  const query = `
    SELECT p.*, c.name AS category_name
    FROM products p
    LEFT JOIN categories c ON p.category_id = c.category_id
    WHERE LOWER(p.name) LIKE LOWER(?)
  `;
  return await db.getAllAsync(query, [`%${name}%`]);
}

// 🆕 Thêm sản phẩm mới
export async function insertProduct(product: Product) {
  await db.runAsync(
    `INSERT INTO products (name, price, category_id, stock)
     VALUES (?, ?, ?, ?)`,
    [product.name, product.price, product.category_id, product.stock ?? 0]
  );
}

// 🧭 Lấy sản phẩm theo danh mục
export async function getProductsByCategory(categoryId: number): Promise<Product[]> {
  return await db.getAllAsync<Product>(
    `SELECT * FROM products WHERE category_id = ? ORDER BY product_id DESC`, [categoryId]
  );
}

export async function deleteProduct(id: number) {
  await db.runAsync("DELETE FROM products WHERE product_id = ?", [id]);
}

export async function updateProduct(product: Product) {
  await db.runAsync(
    "UPDATE products SET name = ?, price = ? WHERE product_id = ?",
    [product.name, product.price, product.product_id]
  );
}
